import fs from 'node:fs/promises';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
import assert from 'node:assert/strict';
import {Workbook,SpreadsheetFile,FileBlob} from '@oai/artifact-tool';
const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'..');
await fs.mkdir(path.join(root,'data'),{recursive:true});await fs.mkdir(path.join(root,'files'),{recursive:true});
const runs=[];
for(const quantity of [2,5,10]){
 const w=Workbook.create();let s;const steps=[];
 const state=()=>({worksheets:w.worksheets.items.map(t=>({name:t.name,range:'A1:D5',values:t.getRange('A1:D5').values,formulas:t.getRange('A1:D5').formulas,tables:t.tables.items.map(x=>({name:x.name,range:x.address})),charts:t.charts.items.map(x=>({type:x.type,title:x.title.text,series:x.series.items.map(v=>({name:v.name,formula:v.formula,categoryFormula:v.categoryFormula}))}))}))});
 async function step(id,label,intent,code,input,fn){const before=state(),returned=await fn();steps.push({id,label,intent,code,input,returned:returned??null,before,after:state()});}
 await step('create','Create','A workbook is the container. A worksheet gives cells a place to live.',"const workbook = Workbook.create();\nconst sheet = workbook.worksheets.add('Sales');",{worksheet:'Sales'},()=>{s=w.worksheets.add('Sales');return {worksheetCount:1};});
 const values=[['Product','Quantity','Price','Revenue'],['Standard',quantity,10,null],['Premium',3,20,null]];
 await step('write','Write','Put the source facts in cells. Revenue is still empty because it is a calculation.',"sheet.getRange('A1:D3').values = input.values;",{range:'A1:D3',values},()=>{s.getRange('A1:D3').values=values;return {writtenRange:'A1:D3'};});
 await step('calculate','Calculate','Store the rule once, fill it down, and calculate the result from the inputs.',"sheet.getRange('D2').formulas = [['=B2*C2']];\nsheet.getRange('D2:D3').fillDown();\nsheet.getRange('C5').values = [['Total']];\nsheet.getRange('D5').formulas = [['=SUM(D2:D3)']];\nworkbook.recalculate();",{formula:'=B2*C2',fill:'D2:D3',total:'=SUM(D2:D3)'},()=>{s.getRange('D2').formulas=[['=B2*C2']];s.getRange('D2:D3').fillDown();s.getRange('C5').values=[['Total']];s.getRange('D5').formulas=[['=SUM(D2:D3)']];w.recalculate();assert.equal(s.getRange('D5').values[0][0],quantity*10+60);return {total:s.getRange('D5').values[0][0]};});
 await step('format','Organize','A named Excel Table adds structure. Formatting helps the reader distinguish inputs and calculated results.',"sheet.tables.add('A1:D3', true, 'SalesData');\nsheet.getRange('C2:D5').setNumberFormat('$#,##0');\nsheet.getRange('A1:D1').format = { fill:'#155e63', font:{bold:true,color:'#ffffff'} };",{table:'SalesData',hasHeaders:true,numberFormat:'$#,##0'},()=>{s.tables.add('A1:D3',true,'SalesData');s.getRange('A1:D5').format.columnWidthPx=112;s.getRange('A1:D5').format.rowHeightPx=28;s.getRange('C2:D5').setNumberFormat('$#,##0');s.getRange('A1:D1').format={fill:'#155e63',font:{bold:true,color:'#ffffff'}};s.showGridLines=false;return {tables:1,format:'currency'};});
 await step('chart','Visualize','The chart points at worksheet ranges. Its stored references explain where the picture gets its values.',"const chart = sheet.charts.add('bar', [sheet.getRange('A1:A3'),sheet.getRange('D1:D3')]);\nchart.setPosition('A8','F19');\nchart.title.text = 'Revenue by product';",{type:'bar',categories:'A1:A3',values:'D1:D3'},()=>{const c=s.charts.add('bar',[s.getRange('A1:A3'),s.getRange('D1:D3')]);c.setPosition('A8','F19');c.title.text='Revenue by product';return {charts:1,source:c.series.items.map(x=>x.formula)};});
 await step('inspect','Inspect','Inspect gives a compact account of the model. Trace follows a result back to the cells that feed it.',"const inspection = await workbook.inspect({kind:'sheet,table,formula,chart'});\nconst trace = workbook.trace('Sales!D5');",{kind:'sheet,table,formula,chart',trace:'Sales!D5'},async()=>({inspection:await w.inspect({kind:'sheet,table,formula,chart',maxChars:16000}),trace:w.trace('Sales!D5')}));
 await step('export','Export','A picture and an Excel workbook are different deliverables. Save both, then check the workbook itself.',"const file = await SpreadsheetFile.exportXlsx(workbook);\nawait file.save('sales.xlsx');\nconst image = await workbook.render({sheetName:'Sales',range:'A1:F19',scale:1.5});",{formats:['xlsx','png']},async()=>{const file=await SpreadsheetFile.exportXlsx(w);await file.save(path.join(root,'files','sales-'+quantity+'.xlsx'));const image=await w.render({sheetName:'Sales',range:'A1:F19',scale:1.5});await fs.writeFile(path.join(root,'files','sales-'+quantity+'.png'),new Uint8Array(await image.arrayBuffer()));return {xlsx:'files/sales-'+quantity+'.xlsx',png:'files/sales-'+quantity+'.png',total:s.getRange('D5').values[0][0]};});
 await step('reopen','Reopen','A saved file deserves its own check. Here the total, formulas, native table and chart survive re-import.',"const reopened = await SpreadsheetFile.importXlsx(await FileBlob.load('sales.xlsx'));\nreopened.recalculate();",{file:'sales.xlsx'},async()=>{const t=await SpreadsheetFile.importXlsx(await FileBlob.load(path.join(root,'files','sales-'+quantity+'.xlsx')));t.recalculate();const z=t.worksheets.getItem('Sales');const result={total:z.getRange('D5').values[0][0],formulas:z.getRange('D2:D3').formulas,tables:z.tables.items.length,charts:z.charts.items.length};assert.equal(result.total,quantity*10+60);assert.equal(result.tables,1);assert.equal(result.charts,1);return result;});
 runs.push({quantity,steps});
}
await fs.writeFile(path.join(root,'data/walkthrough.json'),JSON.stringify({packageVersion:'2.8.59',capturedAt:new Date().toISOString(),kind:'actual package executions with synthetic inputs',runs},null,2));
console.log('Captured 24 steps across three verified input cases.');process.exit(0);
